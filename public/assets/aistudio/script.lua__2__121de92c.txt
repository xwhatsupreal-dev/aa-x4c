local function idx(root, path)
	if not root then return nil end
	local cur = root
	for part in string.gmatch(path, "[^%.]+") do
		if not cur then return nil end
		cur = cur:FindFirstChild(part)
	end
	return cur
end
local function stamp()
	local ok, s = pcall(os.date, "%H:%M:%S")
	return (ok and s) or "--:--:--"
end
local function fmt(n)
	n = tonumber(n) or 0
	if n >= 1e9 then return string.format("%.2fB", n / 1e9) end
	if n >= 1e7 then return string.format("%.1fM", n / 1e6) end
	local s = tostring(math.floor(n))
	local k
	repeat s, k = s:gsub("^(-?%d+)(%d%d%d)", "%1,%2") until k == 0
	return s
end

local function itemImg(id)
	if not id then return nil end
	if OverrideIcons then
		local l = OverrideIcons:FindFirstChild(tostring(id))
		if l then
			for _, x in ipairs(l:GetDescendants()) do
				if (x:IsA("ImageLabel") or x:IsA("ImageButton")) and x.Image ~= "" then return x.Image end
			end
		end
	end
	local d = Items[id]
	if type(d) == "table" then
		for _, k in ipairs({ "icon", "image", "_icon", "_resource_icon" }) do
			local v = d[k]
			if type(v) == "string" and v ~= "" then return v end
		end
	end
	return nil
end

local function mapImage(mapId, world)
	if not MapIcons then return nil end
	for _, key in ipairs({ mapId, world }) do
		if key then
			local f = MapIcons:FindFirstChild(tostring(key))
			if f then
				for _, d in ipairs(f:GetDescendants()) do
					if (d:IsA("ImageLabel") or d:IsA("ImageButton")) and d.Image ~= "" then return d.Image end
				end
			end
		end
	end
	return nil
end

local function readStats()
	local s = LP:FindFirstChild("_stats")
	if not s then return nil end
	local t = { _res = {} }
	for _, c in ipairs(s:GetChildren()) do
		if c:IsA("NumberValue") or c:IsA("IntValue") then
			local v = c.Value
			local nm = c.Name
			local low = nm:lower()
			if low:sub(1, 9) == "_resource" then
				t._res[nm:sub(10)] = v
			elseif low:find("gold") then t.gold = v
			elseif low:find("gem") then t.gem = v
			elseif low:find("xp") then t.xp = v
			elseif low:find("troph") then t.trophy = v
			elseif low:find("token") then t[nm] = v
			end
		end
	end
	return t
end

local function readRes()
	local p = profile()
	local r = p and rawget(p, "resources")
	if type(r) ~= "table" then return {} end
	local out = {}
	for k, v in pairs(r) do
		if type(v) == "number" then out[tostring(k)] = v end
	end
	return out
end

local function snapshot()
	local st = readStats() or {}
	local tokens = {}
	for k, v in pairs(st) do
		if type(k) == "string" and k:lower():find("token") then tokens[k] = v end
	end
	local res = {}
	for k, v in pairs(st._res or {}) do res[k] = v end
	for k, v in pairs(readRes()) do res[k] = (res[k] or 0) + v end
	return {
		gold = st.gold or 0, gem = st.gem or 0, xp = st.xp or 0,
		trophy = st.trophy or 0, tokens = tokens, res = res,
	}
end
H.LastSnap = snapshot()

local function push(label, amount, kind, rar, imgId)
	if not amount or amount == 0 then return end
	local key = kind .. "|" .. tostring(label) .. "|" .. tostring(math.floor(amount * 100)) .. "|" .. stamp()
	if H.Seen[key] then return end
	H.Seen[key] = true
	local e = { kind = kind, label = tostring(label), amount = amount, time = stamp(), rarity = rar, img = imgId }
	table.insert(H.Rewards, 1, e)
	if #H.Rewards > 120 then table.remove(H.Rewards) end
	if H.Flags.Notify and H.NotifyCard then H.NotifyCard(e) end
	if H.RefreshLog then H.RefreshLog() end
end

local function diff()
	local now = snapshot()
	local old = H.LastSnap or now
	if now.gold   > old.gold   then push("Gold",   now.gold   - old.gold,   "cash", "Legendary", "rbxassetid://100751534594369") end
	if now.gem    > old.gem    then push("Gems",   now.gem    - old.gem,    "cash", "Rare",      "rbxassetid://71405874061254") end
	if now.xp     > old.xp     then push("XP",     now.xp     - old.xp,     "cash", "Uncommon",  nil) end
	if now.trophy > old.trophy then push("Trophy", now.trophy - old.trophy, "cash", "Epic",      nil) end
	for k, v in pairs(now.tokens) do
		local ov = old.tokens[k] or 0
		if v > ov then push(k, v - ov, "token", "Epic", itemImg(k)) end
	end
	for k, v in pairs(now.res) do
		local ov = old.res[k] or 0
		if v > ov then push(k, v - ov, "event", "Mythic", itemImg(k)) end
	end
	H.LastSnap = now
	if H.RefreshMoney then H.RefreshMoney() end
end

local function hookRemotes()
	local ep = RS:FindFirstChild("endpoints")
	local s2c = ep and ep:FindFirstChild("server_to_client")
	if not s2c then return end

	local r1 = s2c:FindFirstChild("unit_killed_resource_reward")
	if r1 then
		conn(r1.OnClientEvent:Connect(function(...)
			local a, b = ...
			if type(a) == "string" and type(b) == "number" then
				push(a, b, "event", "Mythic", itemImg(a))
			elseif type(a) == "table" then
				local id = a.item_id or a.resource_id or a.id or a.name or a.resource
				local amt = a.amount or a.value or a.quantity or a.reward
				if type(id) == "string" and type(amt) == "number" then
					push(id, amt, "event", "Mythic", itemImg(id))
				end
			end
		end))
	end

	local r2 = s2c:FindFirstChild("show_drop_message")
	if r2 then
		conn(r2.OnClientEvent:Connect(function(a)
			if type(a) == "table" then
				local id = a.item_id or a.id or a.name
				local amt = a.amount or a.quantity or 1
				if type(id) == "string" then
					local def = Items[id]
					push((type(def) == "table" and def.name) or id, tonumber(amt) or 1, "item",
						(type(def) == "table" and def.rarity) or "Rare", itemImg(id))
				end
			end
		end))
	end

	local r3 = s2c:FindFirstChild("show_unit_and_item_rewards")
	if r3 then
		conn(r3.OnClientEvent:Connect(function(t)
			if type(t) ~= "table" then return end
			for _, list in ipairs({ "units", "items" }) do
				local L = t[list]
				if type(L) == "table" then
					for _, e in pairs(L) do
						local id = (type(e) == "table" and (e.id or e.unit_id or e.item_id)) or e
						if type(id) == "string" then
							local isU = (list == "units")
							local d = isU and Units[id] or Items[id]
							push((type(d) == "table" and d.name) or id, 1,
								isU and "unit" or "item",
								(type(d) == "table" and d.rarity) or "Rare", itemImg(id))
						end
					end
				end
			end
		end))
	end

	local gf = s2c:FindFirstChild("game_finished")
	if gf then
		conn(gf.OnClientEvent:Connect(function()
			task.delay(1.5, diff); task.delay(3.5, diff); task.delay(6, diff)
		end))
	end
end
hookRemotes()
task.spawn(function()
	while H.Running do
		task.wait(2)
		pcall(diff)
	end
end)

local function readModifiers()
	local out = {}
	local ir = idx(PG, "GameLevelInfo.List_rg.inner.scroll.Buffs")
	if ir then
		for _, c in ipairs(ir:GetChildren()) do
			if c:IsA("TextLabel") and c.Text ~= "" and c.Name ~= "InfoReward_title" then
				table.insert(out, c.Text)
			end
		end
	end
	return out
end

local function readLevel()
	local n = idx(PG, "Waves.HealthBar.LevelName")
	local mapName = (n and n.Text ~= "" and n.Text) or nil
	if not mapName then
		local gl = PG:FindFirstChild("GameLevelInfo")
		local t = gl and gl:FindFirstChild("Title", true)
		mapName = t and t.Text
	end
	local mapId = nil
	for id, d in pairs(Maps) do
		if type(d) == "table" then
			local dn = tostring(d.name or ""):lower()
			if mapName and mapName ~= "" and dn ~= ""
				and (dn == mapName:lower() or mapName:lower():find(dn, 1, true)) then
				mapId = id
				break
			end
		end
	end
	local inLobby = LP:FindFirstChild("InLobby")
	local isLobby = (inLobby and inLobby.Value == true) or (mapName == nil)
	return {
		name = mapName or (isLobby and "Lobby" or "?"),
		id = mapId or mapName,
		img = mapImage(mapId or mapName, mapName),
		lobby = isLobby,
		modifiers = readModifiers(),
	}
end

local function teamSlots()
	local out = {}
	local tu = PG:FindFirstChild("PlayerTeam")
	local root = tu and (tu:FindFirstChild("Units") or tu:FindFirstChild("Team") or tu)
	if not root then return out end
	local i = 0
	for _, c in ipairs(root:GetDescendants()) do
		if c:IsA("ImageLabel") or c:IsA("ImageButton") then
			if c.Name:find("Unit") or c.Name:find("Slot") or c:FindFirstChild("UnitName", true) then
				i = i + 1
				if i > 6 then break end
				local lbl = c:FindFirstChild("Name", true) or c:FindFirstChild("UnitName", true)
				out[i] = {
					idx = i,
					name = (lbl and lbl.Text) or "?",
					img = (c.Image ~= "" and c.Image) or nil,
				}
			end
		end
	end
	return out
end

local function refreshTeam()
	local p = profile()
	local col = p and rawget(p, "collection")
	local eq = col and rawget(col, "equipped_units")
	local slots = teamSlots()
	if type(eq) == "table" then
		local list = {}
		for _, v in pairs(eq) do
			if type(v) == "table" and type(v.id) == "string" then
				table.insert(list, v.id)
			elseif type(v) == "string" then
				table.insert(list, v)
			end
		end
		table.sort(list)
		for i = 1, math.min(6, #list) do
			local id = list[i]
			local prev = slots[i] or { idx = i }
			local ud = Units[id]
			slots[i] = {
				idx = i,
				name = (type(ud) == "table" and ud.name) or prev.name or id,
				img = prev.img,
				rarity = type(ud) == "table" and ud.rarity or nil,
			}
		end
	end
	H.Team = slots
end

H.Stop = function()
	H.Running = false
	for _, c in ipairs(H.Conns) do
		pcall(function() c:Disconnect() end)
	end
	if gui then gui:Destroy() end
end

task.spawn(function()
	while H.Running do
		pcall(function()
			local lv = readLevel()
			H.LevelName = lv.name
			H.MapImg = lv.img
			mapNameLbl.Text = (lv.lobby and "Lobby (ยังไม่ลงด่าน)" or lv.name)
				.. ((lv.id and lv.id ~= lv.name) and ("\n[" .. tostring(lv.id) .. "]") or "")
			if lv.img then mapThumb.Image = lv.img end
			modLbl.Text = (#lv.modifiers > 0) and table.concat(lv.modifiers, "\n") or ""
			refreshTeam()
			drawTeam()
		end)
		task.wait(3)
	end
end)
