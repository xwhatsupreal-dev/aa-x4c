local function idx(root, path)
	if not root then return nil end
	local cur = root
	for part in string.gmatch(path, "[^%.]+") do
		if not cur then return nil end
		cur = cur:FindFirstChild(part)
	end
	return cur
end
local function txt(root, path)
	local i = idx(root, path)
	if i and (i:IsA("TextLabel") or i:IsA("TextButton")) then return i.Text end
	return nil
end
local function imgOf(root, path)
	local i = idx(root, path)
	if i and (i:IsA("ImageLabel") or i:IsA("ImageButton")) and i.Image ~= "" then return i.Image end
	return nil
end
local function stamp()
	local ok, s = pcall(os.date, "%H:%M:%S")
	if ok and s then return s end
	return "--:--:--"
end

local function itemImg(id)
	if not id then return nil end
	if OverrideIcons then
		local l = OverrideIcons:FindFirstChild(tostring(id))
		if l then
			for _, x in ipairs(l:GetDescendants()) do
				if (x:IsA("ImageLabel") or x:IsA("ImageButton")) and x.Image ~= "" then return x.Image end
			end
		end
	end
	local d = Items[id]
	if type(d) == "table" then
		for _, k in ipairs({ "icon", "image", "_icon", "_resource_icon" }) do
			local v = d[k]
			if type(v) == "string" and v ~= "" then return v end
		end
	end
	return nil
end

local function itemModel(id)
	if not (ItemModels and id) then return nil end
	local d = Items[id]
	local key = (type(d) == "table" and d._use_item_model) or id
	local m = ItemModels:FindFirstChild(tostring(key))
	if m and (m:IsA("Model") or m:IsA("BasePart")) then return m end
	return nil
end

local function scoreProfile(o)
	local s = 0
	local col = rawget(o, "collection")
	if type(col) == "table" then
		s = s + 1
		if type(rawget(col, "owned_units")) == "table" then s = s + 3 end
		if type(rawget(col, "equipped_units")) == "table" then s = s + 2 end
	end
	local inv = rawget(o, "inventory")
	if type(inv) == "table" then
		s = s + 1
		if type(rawget(inv, "normal_items")) == "table" then s = s + 2 end
	end
	if type(rawget(o, "loadouts_v2")) == "table" then s = s + 2 end
	if type(rawget(o, "gold_amount")) == "number" then s = s + 1 end
	return s
end

local function findProfile()
	if H.Profile and scoreProfile(H.Profile) >= 9 then return H.Profile end
	local best, bestScore = nil, 0
	for _, o in ipairs(getgc(true)) do
		if type(o) == "table" then
			local ok, sc = pcall(scoreProfile, o)
			if ok and sc > bestScore then best, bestScore = o, sc end
		end
	end
	if best and bestScore >= 6 then
		H.Profile = best
		return best
	end
	return nil
end

local function snapStats()
	local t = {}
	local s = LP:FindFirstChild("_stats")
	if s then
		for _, c in ipairs(s:GetChildren()) do
			if c:IsA("ValueBase") then
				local ok, v = pcall(function() return c.Value end)
				if ok and type(v) == "number" then t[c.Name] = v end
			end
		end
	end
	return t
end

local function snapProf()
	local items, units, uniques = {}, {}, {}
	local p = findProfile()
	if p then
		local inv = rawget(p, "inventory")
		if type(inv) == "table" then
			local ni = rawget(inv, "normal_items")
			if type(ni) == "table" then
				for id, amt in pairs(ni) do
					if type(id) == "string" and type(amt) == "number" then items[id] = amt end
				end
			end
			local ui = rawget(inv, "unique_items")
			if type(ui) == "table" then
				for uuid, u in pairs(ui) do
					if type(u) == "table" and type(u.item_id) == "string" then uniques[uuid] = u.item_id end
				end
			end
		end
		local col = rawget(p, "collection")
		if type(col) == "table" then
			local ou = rawget(col, "owned_units")
			if type(ou) == "table" then
				for uuid, u in pairs(ou) do
					if type(u) == "table" and type(u.unit_id) == "string" then units[uuid] = u.unit_id end
				end
			end
		end
	end
	return { items = items, units = units, uniques = uniques }
end

local function makeModelThumb(parent, src, px, xo, yo)
	local vf = Instance.new("ViewportFrame")
	vf.Size = UDim2.fromOffset(px, px)
	vf.Position = UDim2.fromOffset(xo, yo)
	vf.BackgroundColor3 = Color3.fromRGB(38, 38, 50)
	vf.BorderSizePixel = 0
	vf.Ambient = Color3.fromRGB(190, 190, 205)
	vf.LightColor = Color3.fromRGB(255, 255, 255)
	vf.Parent = parent
	Instance.new("UICorner", vf).CornerRadius = UDim.new(0, 6)
	local ok, clone = pcall(function() return src:Clone() end)
	if not ok or not clone then return vf end
	local wm = Instance.new("WorldModel")
	wm.Parent = vf
	clone.Parent = wm
	local center, size
	if clone:IsA("Model") then
		local okc, c, s = pcall(function() return clone:GetBoundingBox() end)
		if okc and c then center, size = c, s end
	else
		center, size = clone.CFrame, clone.Size
	end
	if not center then return vf end
	local md = math.max(size.X, size.Y, size.Z, 0.2)
	local dir = Vector3.new(0.55, 0.35, 0.76).Unit
	local cam = Instance.new("Camera")
	cam.FieldOfView = 40
	cam.CFrame = CFrame.lookAt(center.Position + dir * (md * 2.0 + 0.4), center.Position)
	cam.Parent = vf
	vf.CurrentCamera = cam
	return vf
end

		l.Size = UDim2.fromScale(1, 1)
		l.BackgroundTransparency = 1
		l.Font = Enum.Font.GothamBold
		l.TextSize = math.floor(px * 0.42)
		l.TextColor3 = Color3.fromRGB(255, 255, 255)
		l.Text = string.upper(string.sub(tostring(u.name), 1, 1))
		l.Parent = g
		H.Portraits[u.uuid] = wrap
		return wrap
	end

	local vf = Instance.new("ViewportFrame")
	vf.Size = UDim2.new(1, -2, 1, -2)
	vf.Position = UDim2.fromOffset(1, 1)
	vf.BackgroundColor3 = Color3.fromRGB(34, 34, 46)
	vf.BorderSizePixel = 0
	vf.Ambient = Color3.fromRGB(195, 195, 210)
	vf.LightColor = Color3.fromRGB(255, 255, 255)
	vf.LightDirection = Vector3.new(-0.5, -0.8, -0.6)
	vf.Parent = wrap
	Instance.new("UICorner", vf).CornerRadius = UDim.new(0, 6)

	local ok, clone = pcall(function() return u.model:Clone() end)
	if not ok or not clone then
		H.Portraits[u.uuid] = wrap
		return wrap
	end
	for _, d in ipairs(clone:GetDescendants()) do
		if d:IsA("Sound") or d:IsA("Script") or d:IsA("LocalScript") or d:IsA("ParticleEmitter")
			or d:IsA("Trail") or d:IsA("Beam") or d:IsA("BillboardGui") then
			pcall(function() d:Destroy() end)
		end
	end
	local wm = Instance.new("WorldModel")
	wm.Parent = vf
	clone.Parent = wm

	local head = clone:FindFirstChild("Head", true)
	local target, dir, dist
	if head and head:IsA("BasePart") then
		local lv = head.CFrame.LookVector
		if lv.Magnitude < 0.05 then lv = Vector3.new(0, 0, -1) end
		local hs = math.max(head.Size.X, head.Size.Y, head.Size.Z, 0.5)
		target = head.Position + Vector3.new(0, -hs * 0.35, 0)
		dir = (lv.Unit * 0.94 + Vector3.new(0, 0.22, 0)).Unit
		dist = hs * 3.4
	else
		local okc, c, s = pcall(function() return clone:GetBoundingBox() end)
		if okc and c then
			target = c.Position
			dist = math.max(s.X, s.Y, s.Z) * 2.0
			dir = Vector3.new(0.55, 0.35, 0.76).Unit
		end
	end
	if target and dir and dist then
		local cam = Instance.new("Camera")
		cam.FieldOfView = 36
		cam.CFrame = CFrame.lookAt(target + dir * dist, target)
		cam.Parent = vf
		vf.CurrentCamera = cam
	end

	H.Portraits[u.uuid] = wrap
	return wrap
end

local function levelState()

	if mc then

		if ok and type(d) == "table" then st.raw = d end
	end

local function guiUnitModels()
	local map = {}

local function readTeam()
	local out = {}
	local p = findProfile()
	if not p then return out end
	local col = rawget(p, "collection")
	if type(col) ~= "table" then return out end
	local ou = rawget(col, "owned_units")
	if type(ou) ~= "table" then return out end
	local models = guiUnitModels()
	local ingLv = rawget(col, "unit_ingame_levels")

	local function push(slot, uuid)
		local u = ou[uuid]
		if type(u) ~= "table" then return end
		local d = Units[u.unit_id]
		local lv = "?"
		if type(ingLv) == "table" and type(ingLv[uuid]) == "number" then
			lv = tostring(math.floor(ingLv[uuid]))
		end
		out[slot] = {
			slot = slot, uuid = uuid, id = u.unit_id,
			name = (type(d) == "table" and d.name) or tostring(u.unit_id),
			rarity = (type(d) == "table" and d.rarity) or "Common",
			shiny = u.shiny == true, level = lv,
			model = models[slot],
		}
	end

	local eq = rawget(col, "equipped_units")
	if type(eq) == "table" then
		for slot, uuid in pairs(eq) do
			local sn = tonumber(slot)
			if sn and type(uuid) == "string" then push(sn, uuid) end
		end
	end
	if next(out) == nil then
		for uuid, u in pairs(ou) do
			if type(u) == "table" and type(u.equipped_slot) == "number" then
				push(u.equipped_slot, uuid)
			end
		end
	end
	return out
end

function H.ReadBag()
	local out = {}
	local p = findProfile()
	if not p then return out end
	local inv = rawget(p, "inventory")
	if type(inv) ~= "table" then return out end
	local ni = rawget(inv, "normal_items")
	if type(ni) == "table" then
		for id, amt in pairs(ni) do
			if type(id) == "string" then
				local d = Items[id]
				out[#out + 1] = {
					kind = "item", id = id, qty = amt,
					name = (type(d) == "table" and d.name) or id,
					rarity = (type(d) == "table" and d.rarity) or "Common",
					cat = tostring((type(d) == "table" and d.item_group) or "Other"),
				}
			end
		end
	end
	local ui = rawget(inv, "unique_items")
	if type(ui) == "table" then
		local agg, order = {}, {}
		for _, u in pairs(ui) do
			if type(u) == "table" and type(u.item_id) == "string" then
				if not agg[u.item_id] then
					agg[u.item_id] = 0
					order[#order + 1] = u.item_id
				end
				agg[u.item_id] = agg[u.item_id] + 1
			end
		end
		for _, iid in ipairs(order) do
			local d = Items[iid]
			out[#out + 1] = {
				kind = "special", id = iid, qty = agg[iid],
				name = (type(d) == "table" and d.name) or iid,
				rarity = (type(d) == "table" and d.rarity) or "Common",
				cat = tostring((type(d) == "table" and d.item_group) or "Unique"),
			}
		end
	end
	local col = rawget(p, "collection")
	local ou = type(col) == "table" and rawget(col, "owned_units") or {}
	if type(ou) == "table" then
		local agg, order = {}, {}
		for _, u in pairs(ou) do
			if type(u) == "table" and type(u.unit_id) == "string" then
				local cur = agg[u.unit_id]
				if not cur then
					cur = { n = 0 }
					agg[u.unit_id] = cur
					order[#order + 1] = u.unit_id
				end
				cur.n = cur.n + 1
			end
		end
		for _, uid in ipairs(order) do
			local a = agg[uid]
			local d = Units[uid]
			out[#out + 1] = {
				kind = "unit", id = uid, qty = a.n,
				name = (type(d) == "table" and d.name) or uid,
				rarity = (type(d) == "table" and d.rarity) or "Common",
				cat = "Units", extra = "",
			}
		end
	end
	table.sort(out, function(a, b)
		if a.cat ~= b.cat then return a.cat < b.cat end
		return tostring(a.name) < tostring(b.name)
	end)
	H.Bag = out
	return out
end

local renderQueued = false
local function queueRender()
	if renderQueued then return end
	renderQueued = true
	task.delay(0.35, function()
		renderQueued = false
		pcall(H.Render)
	end)
end

local function addReward(kind, label, amount, rarity, img, extra, id)
	table.insert(H.Rewards, 1, {
		t = stamp(), kind = kind, label = tostring(label), amount = tostring(amount),
		rarity = rarity or "Common", img = img, extra = extra or "", id = id,
		level = H.LevelName or "?",
	})
	while #H.Rewards > 220 do table.remove(H.Rewards) end
	if H.Flags.Notify then
		toast(label, tostring(amount) .. ((extra ~= nil and extra ~= "") and ("  •  " .. tostring(extra)) or ""), rcol(rarity), img)
	end
	queueRender()
end

local function diffStats(old, new)
	if type(old) ~= "table" then return end
	for key, meta in pairs(STAT_KEYS) do
		local o, n = old[key], new[key]
		if type(o) == "number" and type(n) == "number" and n > o then
			local d = n - o
			if d >= 1 then
				addReward("Currency", meta[1], "+" .. tostring(math.floor(d)), meta[2],
					meta[3] and RES_ICON[meta[3]] or nil, "", nil)
			end
		end
	end
end

local function diffProf(old, new)
	if type(old) ~= "table" then return end
	for id, amt in pairs(new.items) do
		if type(amt) == "number" then
			local o = old.items[id]
			local d = o and (amt - o) or amt
			if d and d > 0 then
				local def = Items[id]
				addReward("Item", (type(def) == "table" and def.name) or id, "+" .. tostring(math.floor(d)),
					(type(def) == "table" and def.rarity) or "Common", itemImg(id),
					(type(def) == "table" and def.item_group) or "Item", id)
			end
		end
	end
	for uuid, uid in pairs(new.units) do
		if not old.units[uuid] then
			local def = Units[uid]
			addReward("Unit", (type(def) == "table" and def.name) or uid, "ได้ใหม่!",
				(type(def) == "table" and def.rarity) or "Common", nil, "Unit", uid)
		end
	end
	for uuid, iid in pairs(new.uniques) do
		if not old.uniques[uuid] then
			local def = Items[iid]
			addReward("Special", (type(def) == "table" and def.name) or iid, "ได้ใหม่!",
				(type(def) == "table" and def.rarity) or "Common", itemImg(iid),
				(type(def) == "table" and def.item_group) or "Special", iid)
		end
	end
end

		for _, e in ipairs(H.Bag) do
			order = order + 1
			if e.cat ~= lastCat then
				lastCat = e.cat
				local h = header("▸ " .. tostring(e.cat), Color3.fromRGB(160, 200, 255))
				h.LayoutOrder = order
				order = order + 1
			end
			local r = box(44)
			r.LayoutOrder = order
			local src = nil
			if e.kind ~= "unit" then
				src = itemModel(e.id)
				if src and budget <= 0 then src = nil end
				if src then budget = budget - 1 end
			end
			thumb(r, 34, 5, 5, itemImg(e.id), src, rcol(e.rarity), string.sub(tostring(e.name), 1, 1))
			local sub = tostring(e.rarity) .. (e.kind == "unit" and "" or ("  •  " .. tostring(e.cat)))
			twoLine(r, e.name, sub, rcol(e.rarity), 46, "x" .. tostring(e.qty))
		end
	end
end

local function updateLocation()
	local st = levelState()
	H.State = st
	local d = st.raw
	if st.lobby or not d then
		H.LevelName = "Lobby"
		H.MapImg = mapImage(nil, nil)
		locTitle.Text = "ล็อบบี้"
		locSub.Text = "ยังไม่ได้อยู่ในด่าน\nเลือกด่านจาก Level Select"
	else
		H.LevelName = tostring(d.name)
		H.MapImg = mapImage(d.map, d.world)
		locTitle.Text = tostring(d.name) .. (d.world and ("  •  " .. tostring(d.world)) or "")
		locSub.Text = "แผนที่ " .. tostring(d.map) .. "  •  " .. tostring(d._difficulty or "?") ..
			"\nBoss " .. tostring(d.boss or "-") .. "  •  Wave " .. tostring(st.wave) ..
			(st.enemies and ("  •  เหลือ " .. tostring(st.enemies)) or "")
	end
	locImg.Image = H.MapImg or ""
end

H.LastStats = snapStats()
H.LastProf = snapProf()

task.spawn(function()
	local t = 0
	while H.Running do
		if t % 3 == 0 then pcall(updateLocation) end
		pcall(function()
			local now = snapStats()
			diffStats(H.LastStats, now)
			H.LastStats = now
		end)
		if t % 2 == 0 then
			pcall(function()
				local np = snapProf()
				diffProf(H.LastProf, np)
				H.LastProf = np
			end)
		end
		pcall(captureResults)
		if t % 5 == 0 then
			pcall(function()
				H.Team = readTeam()
				local lv = txt(PG:FindFirstChild("spawn_units"), "Main.Desc.Level")
				footer.Text = string.format("ด่าน: %s   Lv.%s\nGold %s  •  Gems %s  •  XP %s",
					tostring(H.LevelName), tostring(lv or "?"),
					tostring(math.floor(H.LastStats.gold_amount or 0)),
					tostring(math.floor(H.LastStats.gem_amount or 0)),
					tostring(math.floor(H.LastStats.player_xp or 0)))
			end)
		end
		if H.Tab == "level" and t % 9 == 0 then pcall(H.Render) end
		t = t + 1
		task.wait(1)
	end
end)
