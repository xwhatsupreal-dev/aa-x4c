local function idx(root, path)
	if not root then return nil end
	local cur = root
	for part in string.gmatch(path, "[^%.]+") do
		if not cur then return nil end
		cur = cur:FindFirstChild(part)
	end
	return cur
end
local function stamp()
	local ok, s = pcall(os.date, "%H:%M:%S")
	return (ok and s) or "--:--:--"
end
local function fmt(n)
	n = tonumber(n) or 0
	if n >= 1e9 then return string.format("%.2fB", n / 1e9) end
	if n >= 1e7 then return string.format("%.1fM", n / 1e6) end
	local s = tostring(math.floor(n))
	local k
	repeat s, k = s:gsub("^(-?%d+)(%d%d%d)", "%1,%2") until k == 0
	return s
end
local function pretty(k)
	local s = tostring(k):gsub("_", " ")
	s = s:gsub("(%a)([%w]*)", function(a, b) return a:upper() .. b end)
	return s
end

	local d = Items[id]

	local d = Items[id]

	return nil

-- Tier: charm มี field tier, portal ใช้ rarity
local function iTier(id)
	local d = Items[id]
	if type(d) ~= "table" then return nil end
	if d.tier ~= nil then return "Tier " .. tostring(d.tier) end
	if d.rarity ~= nil then return tostring(d.rarity) end

	local r = iRarity(id)

local function iImg(id)
	if not id then return nil end
	if OverrideIcons then
		local l = OverrideIcons:FindFirstChild(tostring(id))
		if l then
			for _, x in ipairs(l:GetDescendants()) do
				if (x:IsA("ImageLabel") or x:IsA("ImageButton")) and x.Image ~= "" then return x.Image end
			end
		end
	end
	local d = Items[id]
	if type(d) == "table" then
		for _, k in ipairs({ "icon", "image", "_icon", "_resource_icon" }) do
			if type(d[k]) == "string" and d[k] ~= "" then return d[k] end
		end
	end
	return nil
end

local function profile()
	if H.P then return H.P end
	local ok, list = pcall(getgc, true)
	if not ok then return nil end
	for _, o in ipairs(list) do
		if type(o) == "table" and type(rawget(o, "inventory")) == "table"
			and type(rawget(o, "level_data")) == "table" then
			H.P = o
			return o
		end
	end
	return nil
end
local function statOf(name)
	local st = LP:FindFirstChild("_stats")
	local v = st and st:FindFirstChild(name)
	if v and v:IsA("ValueBase") then return v.Value end
	return nil
end
local function pScalar(key)
	local P = profile()
	if not P then return nil end
	local v = rawget(P, key)
	if type(v) == "number" or type(v) == "string" then return v end
	return nil
end
local function pResources()
	local out = {}
	local P = profile()
	local r = P and rawget(P, "resources")
	if type(r) == "table" then
		for k, v in pairs(r) do
			if type(v) == "number" and v > 0 then out[tostring(k)] = v end
		end
	end
	local st = LP:FindFirstChild("_stats")
	if st then
		for _, c in ipairs(st:GetChildren()) do
			if c:IsA("ValueBase") and string.sub(c.Name, 1, 9) == "_resource"
				and type(c.Value) == "number" and c.Value > 0 then
				local nm = string.sub(c.Name, 10)
				out[nm] = math.max(out[nm] or 0, c.Value)
			end
		end
	end
	return out
end

local function push(block)

end

local NUMTAGS = { "gold", "gem", "troph", "xp", "resource", "token", "pearl", "star", "point", "currency" }
local function extractNums(t, out, seen, depth, budget)
	if type(t) ~= "table" or depth > 4 then return end
	for k, v in pairs(t) do
		budget.n = budget.n + 1
		if budget.n > 900 then return end
		if type(k) == "string" then
			if type(v) == "number" then
				if v > 0 then
					local kl = k:lower()
					for _, tag in ipairs(NUMTAGS) do
						if string.find(kl, tag, 1, true) then
							if not seen[k] then
								seen[k] = true
								out[#out + 1] = { key = k, val = v }
							end
							break
						end
					end
				end
			elseif type(v) == "table" then
				local isMap, cnt = true, 0
				for k2, v2 in pairs(v) do
					cnt = cnt + 1
					if type(k2) ~= "string" or type(v2) ~= "number" then isMap = false; break end
				end
				if isMap and cnt > 0 then
					for k2, v2 in pairs(v) do
						if v2 > 0 and not seen[k2] then
							seen[k2] = true
							out[#out + 1] = { key = k2, val = v2, parent = k }
						end
					end
				else
					extractNums(v, out, seen, depth + 1, budget)
				end
			end
		end
	end
end

local function colorFor(key)
	local kl = string.lower(tostring(key))
	if string.find(kl, "gold", 1, true) then return C.gold end
	if string.find(kl, "gem", 1, true) then return C.gem end
	if string.find(kl, "xp", 1, true) then return C.xp end
	if string.find(kl, "troph", 1, true) then return C.acc end
	return C.acc
end

local function onFinished(payload)
	H.LastFinish = payload
	H.Raw[#H.Raw + 1] = { tag = "game_finished", t = stamp(), data = payload }
	while #H.Raw > 12 do table.remove(H.Raw, 1) end
	if type(payload) ~= "table" then return end

	local rows, seen = {}, {}
	extractNums(payload, rows, seen, 0, { n = 0 })
	table.sort(rows, function(a, b) return a.val > b.val end)

	local conv = {}
	for _, r in ipairs(rows) do
		conv[#conv + 1] = {
			text = "+" .. fmt(r.val) .. "  " .. pretty(r.key),
			color = colorFor(r.key),
			icon = RES_ICON[r.key],
		}
	end

	local st = payload.stats
	if type(st) == "table" then
		if st.waves_completed then conv[#conv + 1] = { text = "Waves: " .. tostring(st.waves_completed), color = C.sub } end
		if st.time_taken then
			local t = math.floor(tonumber(st.time_taken) or 0)
			conv[#conv + 1] = { text = string.format("Time: %02d:%02d", math.floor(t / 60), t % 60), color = C.sub }
		end
	end

	local win = payload.victory
	if win == nil and payload.title then win = not string.find(string.lower(tostring(payload.title)), "fail", 1, true) end
	push({
		title = (win and "VICTORY" or "DEFEAT") .. " — " .. tostring(payload.title or payload.level or "?"),
		color = win and C.good or C.bad,
		rows = conv,
	})
	if H.renderLevel then pcall(H.renderLevel) end
end

-- ดึง item id + จำนวน จากอากิวเมนต์แบบไม่ต้องรู้ลำดับ
local function idsFromPacked(packed)
	local found = {}
	local function add(id, n)
		if type(id) ~= "string" or not Items[id] then return end
		for _, f in ipairs(found) do
			if f.id == id then f.n = f.n + (tonumber(n) or 1); return end
		end
		found[#found + 1] = { id = id, n = tonumber(n) or 1 }
	end
	local function rec(v, depth)
		if depth > 3 then return end
		local tv = type(v)
		if tv == "string" then
			add(v, 1)
		elseif tv == "number" then
			if #found > 0 then found[#found].n = v end
		elseif tv == "table" then
			local id = rawget(v, "item_id") or rawget(v, "id")
			if type(id) == "string" then
				add(id, rawget(v, "amount") or rawget(v, "quantity") or 1)
			end
			for k, val in pairs(v) do
				if type(k) == "string" and Items[k] and type(val) == "number" then
					add(k, val)
				elseif type(val) == "table" or type(val) == "string" then
					rec(val, depth + 1)
				end
			end
		end
	end
	for i = 1, packed.n do rec(packed[i], 0) end
	return found
end

local function onItems(packed)
	H.Raw[#H.Raw + 1] = { tag = "show_unit_and_item_rewards", t = stamp(), data = packed }
	while #H.Raw > 12 do table.remove(H.Raw, 1) end
	local items = idsFromPacked(packed)
	local rows = {}
	for _, it in ipairs(items) do
		local tier = iTier(it.id)
		rows[#rows + 1] = {
			text = "+" .. fmt(it.n) .. "  " .. iName(it.id) .. (tier and ("   [" .. tier .. "]") or ""),
			color = iColor(it.id),
			icon = iImg(it.id),
		}
	end
	local function countUnits(v, depth)
		if depth > 3 or type(v) ~= "table" then return end
		for k, val in pairs(v) do
			if type(k) == "string" and Units[k] then
				rows[#rows + 1] = {
					text = "+ ยูนิต: " .. tostring((type(Units[k]) == "table" and Units[k].name) or k),
					color = Color3.fromRGB(255,215,120),
				}
			elseif type(val) == "table" then
				countUnits(val, depth + 1)
			elseif type(val) == "string" and Units[val] then
				rows[#rows + 1] = {
					text = "+ ยูนิต: " .. tostring((type(Units[val]) == "table" and Units[val].name) or val),
					color = Color3.fromRGB(255,215,120),
				}
			end
		end
	end
	countUnits(packed, 0)
	if #rows > 0 then
		push({ title = "ของ / ยูนิตที่ได้รับ", color = C.acc, rows = rows })
	end
end

local function onDrop(packed)
	H.Raw[#H.Raw + 1] = { tag = "show_drop_message", t = stamp(), data = packed }
	while #H.Raw > 12 do table.remove(H.Raw, 1) end
	local items = idsFromPacked(packed)
	local rows = {}
	for _, it in ipairs(items) do
		local tier = iTier(it.id)
		rows[#rows + 1] = {
			text = "+" .. fmt(it.n) .. "  " .. iName(it.id) .. (tier and ("   [" .. tier .. "]") or ""),
			color = iColor(it.id),
			icon = iImg(it.id),
		}
	end
	if #rows == 0 then
		local parts = {}
		for i = 1, math.min(packed.n, 4) do
			if type(packed[i]) ~= "table" then parts[#parts + 1] = tostring(packed[i]) end
		end
		rows[1] = { text = table.concat(parts, "  ·  "):sub(1, 90), color = C.sub }
	end
	local head = items[1] and ("ดรอป: " .. iName(items[1].id)) or "ดรอป"
	push({ title = head, color = items[1] and iColor(items[1].id) or C.sub, rows = rows })
end

local function hookAll()
	local ep = idx(RS, "endpoints")
	local s2c = ep and ep:FindFirstChild("server_to_client")
	if not s2c then print("[2K] no server_to_client") return end

	local function hook(name, handler)
		local r = s2c:FindFirstChild(name)
		if r and r:IsA("RemoteEvent") then
			conn(r.OnClientEvent:Connect(function(...)
				local packed = table.pack(...)
				pcall(handler, packed)
			end))
			print("[2K] hooked " .. name)
		end
	end
	hook("game_finished", function(p) onFinished(p[1] or p[2]) end)
	hook("show_unit_and_item_rewards", onItems)
	hook("show_drop_message", onDrop)
	hook("show_drop_message_custom", onDrop)
end

local function new(cls, props, parent)
	local o = Instance.new(cls)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then o[k] = v end
	end
	if parent then o.Parent = parent end
	return o
end
local function corner(o, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r or 8)
	c.Parent = o
	return c
end
local function stroke(o, col, th)
	local s = Instance.new("UIStroke")
	s.Color = col or C.stroke
	s.Thickness = th or 1
	s.Parent = o
	return s
end

local function findKey(t, keyname, depth, budget)
	if type(t) ~= "table" or depth > 4 then return nil end
	for k, v in pairs(t) do
		budget.n = budget.n + 1
		if budget.n > 2500 then return nil end
		if type(k) == "string" and string.lower(k) == keyname and type(v) == "table" then return v end
	end
	for k, v in pairs(t) do
		if type(v) == "table" and type(k) == "string" then
			local r = findKey(v, keyname, depth + 1, budget)
			if r then return r end
		end
	end
	return nil
end
local function teamUnits()
	local P = profile()
	if not P then return {} end
	local t = findKey(P, "team", 0, { n = 0 }) or findKey(P, "equipped_units", 0, { n = 0 })
	local out = {}
	if type(t) ~= "table" then return out end
	for k, v in pairs(t) do
		local uid
		if type(v) == "string" then uid = v
		elseif type(v) == "table" then uid = rawget(v, "unit_id") or rawget(v, "id") end
		if type(uid) == "string" then
			out[#out + 1] = { slot = type(k) == "number" and k or (#out + 1), unit_id = uid }
		end
	end
	table.sort(out, function(a, b) return a.slot < b.slot end)
	return out
end
local function unitIcon(uid)
	local d = Units[uid]
	if type(d) == "table" then
		for _, k in ipairs({ "icon", "image", "_icon" }) do
			if type(d[k]) == "string" and d[k] ~= "" then return d[k] end
		end
	end
	return nil
end

local function scanCounts(t, out, depth, budget)
	if type(t) ~= "table" or depth > 3 then return end
	for k, v in pairs(t) do
		budget.n = budget.n + 1
		if budget.n > 4000 then return end
		if type(v) == "number" and type(k) == "string" and Items[k] then
			out[k] = (out[k] or 0) + v
		elseif type(v) == "table" then
			local id = rawget(v, "item_id") or rawget(v, "id")
			if type(id) == "string" and Items[id] and type(rawget(v, "amount")) == "number" then
				out[id] = (out[id] or 0) + rawget(v, "amount")
			else
				scanCounts(v, out, depth + 1, budget)
			end
		end
	end
end
local function bagCounts()
	local out = {}
	local P = profile()
	local inv = P and rawget(P, "inventory")
	if type(inv) ~= "table" then return out end
	scanCounts(inv, out, 0, { n = 0 })
	return out
end

function H.renderBag()
	clearPage("bag")
	local counts = bagCounts()
	local byGroup = {}
	for id, cnt in pairs(counts) do
		local g = iGroup(id)
		local tier = iTier(id) or "ไม่ระบุ Tier"
		byGroup[g] = byGroup[g] or {}
		byGroup[g][tier] = byGroup[g][tier] or {}
		table.insert(byGroup[g][tier], { id = id, n = cnt })
	end
	local gnames = {}
	for g in pairs(byGroup) do gnames[#gnames + 1] = g end
	table.sort(gnames)
	if #gnames == 0 then
		local s = section("bag", "กระเป๋า", 1)
		label(s, "อ่านกระเป๋าไม่ได้ / ว่าง", C.sub, 1)
		return
	end
	local order = 1
	for _, g in ipairs(gnames) do
		local s = section("bag", g, order)
		order = order + 1
		local tiers = {}
		for t in pairs(byGroup[g]) do tiers[#tiers + 1] = t end
		table.sort(tiers)
		local lo = 1
		for _, t in ipairs(tiers) do
			local list = byGroup[g][t]
			table.sort(list, function(a, b) return iName(a.id) < iName(b.id) end)
			new("TextLabel", {
				Size = UDim2.new(1, 0, 0, 14), BackgroundTransparency = 1, LayoutOrder = lo,
				Font = Enum.Font.GothamBold, TextSize = 10, TextColor3 = rcol(t), TextXAlignment = Enum.TextXAlignment.Left,
				Text = "— " .. t .. "  (" .. #list .. ") —",
			}, s)
			lo = lo + 1
			for _, e in ipairs(list) do
				row(s, iName(e.id) .. "   x" .. fmt(e.n), iColor(e.id), iImg(e.id), lo)
				lo = lo + 1
			end
		end
	end
end

function H.Stop()
	H.Running = false
	for _, c in ipairs(H.Conns) do pcall(function() c:Disconnect() end) end
	H.Conns = {}
	pcall(function() gui:Destroy() end)
end

task.spawn(function()
	local tick = 0
	while H.Running do
		tick = tick + 1
		local ok, err = pcall(function()
			if H.Tab == "level" then H.renderLevel() end
			if H.Tab == "money" and tick % 2 == 0 then H.renderMoney() end
			if H.Tab == "team" and tick % 4 == 0 then H.renderTeam() end
			if H.Tab == "bag" and tick % 6 == 0 then H.renderBag() end
		end)
		if not ok then warn("[2K] loop: " .. tostring(err)) end
		task.wait(1)
	end
end)
